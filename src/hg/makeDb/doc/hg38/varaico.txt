# Oct 13 2026 - Johannes: release 4 (20261006), adds hs1
# From this release on, dataVersion is read from /gbdb/$D/bbi/varaicoVersion.txt
# (written below), so future data-only updates also update the version shown.
export AMELIE2_CONFIG_JSON=/hive/users/jbirgmei/amelie2/src/scripts/config.json

# build hg19, hg38 and hs1. hs1 maps raw-coordinate variants from papers with the
# UCSC hg19ToHs1/hg38ToHs1 chains (amelie2 asset orig_hs1_liftover_chains):
/cluster/home/jbirgmei/opt_rocky9/miniconda3/envs/amelie2/bin/amelie2 workflow final_out --partition hg19,hg38,hs1

# stage the release (per-assembly subdirs; the .sqlite3 files are for varaico.com):
mkdir /hive/users/jbirgmei/TRACK_VERSIONS/v6_20261006
cp -al /hive/users/jbirgmei/amelie2/data/out/None/full/{hg19,hg38,hs1} /hive/users/jbirgmei/TRACK_VERSIONS/v6_20261006/

# archive the previous release (release 3, "release 3 (20260527)"); hs1 had no
# public release, so only hg19 and hg38:
for db in hg19 hg38; do
    mkdir /hive/data/genomes/$db/bed/varaico.v5_20260527
    cp /hive/data/genomes/$db/bed/varaico/varaico.bb \
       /hive/data/genomes/$db/bed/varaico.v5_20260527/$db.varaico.v5_20260527.bb
    cp /hive/data/genomes/$db/bed/varaico/varaicoSuppl.bb \
       /hive/data/genomes/$db/bed/varaico.v5_20260527/$db.varaicoSuppl.v5_20260527.bb
    mkdir -p /usr/local/apache/htdocs-hgdownload/goldenPath/archive/$db/varaico/v5_20260527
    ln -s /hive/data/genomes/$db/bed/varaico.v5_20260527/$db.varaico.v5_20260527.bb \
          /usr/local/apache/htdocs-hgdownload/goldenPath/archive/$db/varaico/v5_20260527/$db.varaico.v5_20260527.bb
    ln -s /hive/data/genomes/$db/bed/varaico.v5_20260527/$db.varaicoSuppl.v5_20260527.bb \
          /usr/local/apache/htdocs-hgdownload/goldenPath/archive/$db/varaico/v5_20260527/$db.varaicoSuppl.v5_20260527.bb
done

# copy the bigBeds and write the version file:
for db in hg19 hg38 hs1; do
    mkdir -p /hive/data/genomes/$db/bed/varaico
    cp /hive/users/jbirgmei/TRACK_VERSIONS/v6_20261006/$db/normalized_variants.$db.bb \
       /hive/data/genomes/$db/bed/varaico/varaico.bb
    cp /hive/users/jbirgmei/TRACK_VERSIONS/v6_20261006/$db/normalized_variants_supplements.$db.bb \
       /hive/data/genomes/$db/bed/varaico/varaicoSuppl.bb
    echo "release 4 (20261006)" > /hive/data/genomes/$db/bed/varaico/varaicoVersion.txt
done

# finally, (re)point the /gbdb symlinks; hs1's used to point at the
# normalized_variants*.hs1.bb names, now it matches hg19/hg38:
for db in hg19 hg38 hs1; do
    for f in varaico.bb varaicoSuppl.bb varaicoVersion.txt; do
        ln -sfn /hive/data/genomes/$db/bed/varaico/$f /gbdb/$db/bbi/$f
    done
done
