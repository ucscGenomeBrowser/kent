# 2026-11-11 Claude/max: Cancer Mutations collection, Cancer Mutation Atlas (mutAtlas, mutAtlasLolly)

# Cancer Mutation Atlas (CMA), Karchin Lab, JHU, https://cancermutationatlas.org/
# version 2026.09.15 (beta1). Ensembl 115 GFF3 for bcftools csq, OncoTree 2025_10_03 for type names.
mkdir -p /hive/data/genomes/hg38/bed/cancerVariants/mutAtlas

~/kent/src/hg/makeDb/scripts/cancerMutations/mutAtlasDownload.sh 2026.09.15

# atlas.tsv -> VCF -> bcftools csq -> bed9+25 -> mutAtlas.bb, ~4 minutes.
# Also writes mutAtlasFilters.ra, the filterValues lines for cancerMutations.ra
~/kent/src/hg/makeDb/scripts/cancerMutations/mutAtlasBuild.sh
# 3918034 variants written: all rows of atlas.tsv, none skipped
# SNV 3610054, deletion 190553, insertion 74173, MNV 31477, delins 11777
# 85555 variants are outside all Ensembl transcripts ("intergenic")

# mutAtlas (bigBed) and mutAtlasLolly (bigLolly) use the same file
mkdir -p /gbdb/hg38/cancerMutations
ln -s /hive/data/genomes/hg38/bed/cancerVariants/mutAtlas/mutAtlas.bb /gbdb/hg38/cancerMutations/

##############################################################################
# 2026-11-22 Claude/max: Cancer Hotspots (cancerHotspots), cancerhotspots.org, refs #38687
# Hotspots v2 (Chang et al. 2017, PMID 29247016) + the 164 new v3 hotspots
# (Bandlamudi et al. 2026, PMID 41895280). Data license: ODbL.
# The script downloads hotspots_v2.xls, hotspots_v3.xlsx, the v2 MAF (only used to
# get the Ensembl 75 transcript per gene) and the Ensembl 115 GFF3, then places each
# hotspot residue on an Ensembl 115 transcript, checking the reference amino acid and,
# for v2, the hg19 mutation positions from the table lifted to hg38. Needs uv.
mkdir -p /hive/data/genomes/hg38/bed/cancerVariants/cancerHotspots
cd /hive/data/genomes/hg38/bed/cancerVariants/cancerHotspots
~/kent/src/hg/makeDb/scripts/cancerMutations/cancerHotspotsBuild.sh hg38
# hotspots in input: 1329 (v2 single residue/splice 1110, v2 indel 55, v3 164)
# features written: 1328
# not written: RECQL4 X766 splice, its only hg19 position does not lift to hg38
# 9 v2 hotspots have a different residue number on the current transcript
# (MYD88 L265 = L252 on MANE, also FGFR1 and ARID1B), see cancerHotspots.log
# Warnings and transcript choices are in cancerHotspots.log

ln -s /hive/data/genomes/hg38/bed/cancerVariants/cancerHotspots/cancerHotspots.bb /gbdb/hg38/cancerMutations/

##############################################################################
# 2026-11-22 Claude/max: Cancer Hotspots on hg19, refs #38687
# Same script, Ensembl GRCh37 release 87 GFF3, the v2 hg19 positions are used as they are.
# The cancerhotspots.org downloads are shared with the hg38 build.
mkdir -p /hive/data/genomes/hg19/bed/cancerVariants/cancerHotspots/orig
cd /hive/data/genomes/hg19/bed/cancerVariants/cancerHotspots
for f in hotspots_v2.xls hotspots_v3.xlsx cancerhotspots.v2.maf.gz; do
    ln -s /hive/data/genomes/hg38/bed/cancerVariants/cancerHotspots/orig/$f orig/; done
~/kent/src/hg/makeDb/scripts/cancerMutations/cancerHotspotsBuild.sh hg19
# hotspots in input: 1329, features written: 1329, no residue number shifts
ln -s /hive/data/genomes/hg19/bed/cancerVariants/cancerHotspots/cancerHotspots.bb /gbdb/hg19/cancerMutations/

# 2026-11-22 Claude/max: rebuild of the TCGA tracks pancan33 and tcgaBrca, refs #38687
# The 2019 files (/hive/data/outside/gdc/pancan33-variants/maf/pancan33.bb,
# /hive/data/genomes/hg38/bed/cancerVariants/BRCA.bb, #22742) came from an early version of
# cancerMafToBigBed and had zero-length items starting at the 1-based MAF position.
# The converter is now python3 and writes start = MAF Start_Position - 1. Same input:
# GDC data release 10.0 MuTect2 MAFs and the clinical/exposure tables of Jan 2019.
M=/hive/data/outside/gdc/pancan33-variants
mkdir -p /hive/data/genomes/hg38/bed/cancerVariants/tcga/brcaMaf
cd /hive/data/genomes/hg38/bed/cancerVariants/tcga
ln -s $M/maf/995c0111-d90b-4140-bee7-3845436c3b42/TCGA.BRCA.*.maf.gz brcaMaf/
~/kent/src/utils/cancerMafToBigBed brcaMaf $M/annotations/clinical.tsv $M/annotations/exposure.tsv BRCA.bb
~/kent/src/utils/cancerMafToBigBed $M/maf $M/annotations/clinical.tsv $M/annotations/exposure.tsv pancan33.bb
# BRCA 119684 and pancan33 2864792 variants, same counts as in 2019.
# 300 random SNVs of pancan33: reference allele matches hg38 at all 300.
ln -s /hive/data/genomes/hg38/bed/cancerVariants/tcga/pancan33.bb /gbdb/hg38/cancerMutations/pancan33.bb
ln -s /hive/data/genomes/hg38/bed/cancerVariants/tcga/BRCA.bb /gbdb/hg38/cancerMutations/tcgaBrca.bb
