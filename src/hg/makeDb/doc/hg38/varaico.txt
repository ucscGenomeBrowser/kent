# Varaico (tracks varaico, varaicoSuppl) on hg19, hg38 and hs1 - Johannes Birgmeier
#
# Builds are identified by their build date. Data-only updates are installed by
# an OTTO job (src/hg/utils/otto/varaico/, see its README) and pushed to the RR by
# autoPush (#38546), so a new build needs no trackDb change and no push request.
# dataVersion reads /gbdb/$D/bbi/varaicoVersion.txt, which the OTTO job writes.

##############################################################################
# Building and releasing a new build

# 1. build the bigBeds for all three assemblies with the amelie2 pipeline. hs1
# maps variants that papers report only as raw coordinates with the UCSC
# hg19ToHs1/hg38ToHs1 chains (amelie2 asset orig_hs1_liftover_chains):
export AMELIE2_CONFIG_JSON=/hive/users/jbirgmei/amelie2/src/scripts/config.json
/cluster/home/jbirgmei/opt_rocky9/miniconda3/envs/amelie2/bin/amelie2 workflow final_out --partition hg19,hg38,hs1

# 2. stage the build under its date (per-assembly subdirs; the .sqlite3 and
# .vcf.gz files there are for varaico.com, only the .bb files go to the browser):
DATE=$(date +%Y%m%d)
mkdir /hive/users/jbirgmei/TRACK_VERSIONS/$DATE
cp -al /hive/users/jbirgmei/amelie2/data/out/None/full/{hg19,hg38,hs1} /hive/users/jbirgmei/TRACK_VERSIONS/$DATE/

# 3. mark it ready, last. READY holds the version shown on the track page:
echo "$DATE" > /hive/users/jbirgmei/TRACK_VERSIONS/$DATE/READY

# 4. the OTTO job (daily, /hive/data/outside/otto/varaico/doUpdate.sh) then
#   - checks every bigBed (opens, item count within +-20% of the installed file,
#     chroms in chrom.sizes) and refuses the whole build if one fails;
#   - archives the installed build under its date:
#       /hive/data/genomes/$db/bed/varaico.<date>/$db.varaico{,Suppl}.<date>.bb,
#       linked from /usr/local/apache/htdocs-hgdownload/goldenPath/archive/$db/varaico/<date>/
#   - installs the new files as /hive/data/genomes/$db/bed/varaico/varaico{,Suppl}.bb,
#     writes varaicoVersion.txt there, and points /gbdb/$db/bbi/ at all three.
# Preview the checks without changing anything:
#   /hive/data/outside/otto/varaico/doUpdate.sh --dry-run
# autoPush then copies /gbdb/{hg19,hg38,hs1}/bbi/varaico{,Suppl}.bb and
# varaicoVersion.txt to the RR.

# If the OTTO job refuses an intended big change (item count off by more than
# 20%), install by hand with the same steps:
CUR=20261019   # date of the installed build (see currentRelease.txt in the OTTO dir)
NEW=$DATE
for db in hg19 hg38 hs1; do
    bed=/hive/data/genomes/$db/bed/varaico
    mkdir -p /hive/data/genomes/$db/bed/varaico.$CUR \
             /usr/local/apache/htdocs-hgdownload/goldenPath/archive/$db/varaico/$CUR
    for t in varaico varaicoSuppl; do
        cp -p $bed/$t.bb /hive/data/genomes/$db/bed/varaico.$CUR/$db.$t.$CUR.bb
        ln -sfn /hive/data/genomes/$db/bed/varaico.$CUR/$db.$t.$CUR.bb \
                /usr/local/apache/htdocs-hgdownload/goldenPath/archive/$db/varaico/$CUR/$db.$t.$CUR.bb
    done
    cp /hive/users/jbirgmei/TRACK_VERSIONS/$NEW/$db/normalized_variants.$db.bb $bed/.varaico.bb.tmp
    cp /hive/users/jbirgmei/TRACK_VERSIONS/$NEW/$db/normalized_variants_supplements.$db.bb $bed/.varaicoSuppl.bb.tmp
    mv -f $bed/.varaico.bb.tmp $bed/varaico.bb
    mv -f $bed/.varaicoSuppl.bb.tmp $bed/varaicoSuppl.bb
    echo "$NEW" > $bed/varaicoVersion.txt
    for f in varaico.bb varaicoSuppl.bb varaicoVersion.txt; do
        ln -sfn $bed/$f /gbdb/$db/bbi/$f
    done
done
echo "$NEW" > /hive/data/outside/otto/varaico/currentRelease.txt
