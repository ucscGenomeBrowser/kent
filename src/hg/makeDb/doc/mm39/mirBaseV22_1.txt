# 2026-10-15 Claude atanzer: miRBase 22.1 microRNA precursors, mm39, #38548

# Mouse microRNA precursors from miRBase release 22.1 (https://mirbase.org/,
# Kozomara et al. 2019, PMID 30423142), in the Non-coding RNA collection
# (nonCodingRNAs, shared with mm10 in trackDb/mouse/). The miRBase mouse GFF is on
# GRCm38 (mm10), so the mm10 file (doc/mm10/mirBaseV22_1.txt) is lifted to mm39, and the
# track is then built as on mm10 and hg38 (see doc/hg38/mirBaseV22_1.txt).
# Scripts: ~/kent/src/hg/makeDb/scripts/mirBase/

##############################################################################
# miRBase 22.1 (DONE 2026-10-15)
##############################################################################

mkdir -p /hive/data/genomes/mm39/bed/mirBaseV22_1/download
cd /hive/data/genomes/mm39/bed/mirBaseV22_1/download
# Each precursor is lifted together with the matures inside it; it is kept only if
# all of them map, to the chromosome and strand of the precursor, with unchanged
# lengths:
python3 ~/kent/src/hg/makeDb/scripts/mirBase/mirBaseLift.py \
    /hive/data/genomes/mm10/bed/mirBaseV22_1/download/mmu.gff3 \
    /hive/data/genomes/mm10/bed/liftOver/mm10ToMm39.over.chain.gz mmu.gff3
#   1227 precursor loci, 1227 lifted, 0 left out
# download/mmu.gff3 here is this lifted copy (mm39 coordinates), not a miRBase file;
# the miRBase original is in the mm10 download directory. Its header says so, and
# the genome-build lines (GRCm38) are marked as the source assembly. The name stays
# mmu.gff3 because mirBaseBuild.sh reads download/<species>.gff3.

cd /hive/data/genomes/mm39/bed/mirBaseV22_1
# The MirGeneDB ID on the details page is taken from the MirGeneDB track of this
# assembly (doc/mm39/mirGeneDbV3_0.txt, build it first): the MirGeneDB genes whose
# miRBase accession, as MirGeneDB gives it, is this precursor's accession. Nothing is
# inferred from positions; see the notes on MirGeneDB paralog swaps in
# doc/mm39/mirGeneDbV3_0.txt.
bash ~/kent/src/hg/makeDb/scripts/mirBase/mirBaseBuild.sh mm39 mmu
#   1227 precursor sequences
#   1227 precursors, 2110 matures, 1227 folds
#   1227 items, 74 not one hairpin and 49 with an arm in the hairpin loop in the MFE: 0 refolded with the flanks unpaired, 117 as suboptimal hairpins, all with the arms in the stem; 4 with an arm in the loop and 2 with no hairpin within 8 kcal/mol
#   arm in the hairpin loop: MI0001730
#   arm in the hairpin loop: MI0002403
#   arm in the hairpin loop: MI0022890
#   arm in the hairpin loop: MI0023710
#   no hairpin, MFE kept: MI0006300
#   no hairpin, MFE kept: MI0006305
#   structure chosen by hand: MI0006279, -32.70 kcal/mol
#   MI0014099 has 2 matures on the 3p arm
#   MI0014099 has 2 matures on the 5p arm
#   matures per arm: 3p 1062, 5p 1048

bigBedInfo mirBase.bb | grep itemCount
#   itemCount: 1,227
bigBedInfo mirBaseArms.bb | grep itemCount
#   itemCount: 2,110

# Link the files into /gbdb:
mkdir -p /gbdb/mm39/mirBaseV22_1
for f in mirBase.bb mirBaseArms.bb mirBase.ix mirBase.ixx; do
    ln -sf `pwd`/$f /gbdb/mm39/mirBaseV22_1/$f
done

# Checks done on the output:
#   - every folded sequence equals the mm39 sequence at the precursor coordinates
#     (strand-aware), and every armPositions range cuts exactly the mature miRNA's
#     mm39 sequence out of it
#   - sequences, structures, foldType and matures of all precursors are identical to
#     the mm10 track
